```python
import requests

def get_planet_info(url):
    response = requests.get(url)
    data = response.json()
    
    planet_name = data["name"]
    planet_terrain = data["terrain"]
    planet_population = data["population"]
    
    return pd.Series([url, planet_name, planet_terrain, planet_population])

```
