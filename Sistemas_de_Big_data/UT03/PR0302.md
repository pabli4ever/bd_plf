```python
import json
import pandas as pd

with open('data/ventas_este_v2.json', 'r', encoding='utf-8') as f:
    data = json.load(f)

registros = []

for item in data:
    if 'data' not in item:
        continue
    id_transaccion = item['data']['id_registro']
    fecha = pd.to_datetime(
        item['data']['payload']['fecha_evento']
    )
    ciudad = item['data']['payload']['comprador']['ubicacion']['ciudad']
    producto = item['data']['payload']['transaccion']['detalles_producto']['nombre_comercial']
    cantidad = item['data']['payload']['transaccion']['cantidad_comprada']
    precio_lista = item['data']['payload']['transaccion']['detalles_producto']['precio_lista']
    monto_iva = item['data']['payload']['transaccion']['detalles_producto']['impuestos']['monto_iva']

    total_venta = (precio_lista * cantidad) + monto_iva
    
    registros.append({
        'id_transaccion': id_transaccion,
        'fecha': fecha,
        'region': 'Este',
        'producto': producto,
        'cantidad': cantidad,
        'total_venta': total_venta,
        'ciudad': ciudad
    })
```


```python
df_este = pd.DataFrame(registros)
```
